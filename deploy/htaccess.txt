# 10Projects.com — cPanel Routing
# WordPress PHP handles all frontend pages + REST API
#
# WordPress handles: all pages, wp-admin, wp-json, wp-login, sitemaps, cron

# ==================================================================
# SECURITY HEADERS
# ==================================================================
<IfModule mod_headers.c>
    # Prevent clickjacking — only allow our own domain to iframe
    Header always set X-Frame-Options "SAMEORIGIN"

    # Prevent MIME-type sniffing
    Header always set X-Content-Type-Options "nosniff"

    # Referrer policy
    Header always set Referrer-Policy "strict-origin-when-cross-origin"

    # Permissions policy — disable unused APIs
    Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(self), payment=()"

    # Content Security Policy
    Header always set Content-Security-Policy "default-src 'self'; script-src 'self' 'unsafe-inline' https://www.googletagmanager.com https://www.google-analytics.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src 'self' https://fonts.gstatic.com; img-src 'self' data: https: blob:; connect-src 'self' https://leadmax.siteclap.com https://www.google-analytics.com https://region1.google-analytics.com; frame-ancestors 'self';"

    # Cache control for static assets
    <FilesMatch "\.(js|css|woff2?|ttf|eot|svg|png|jpg|jpeg|webp|avif|ico|gif)$">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </FilesMatch>
</IfModule>

# ==================================================================
# HOTLINK PROTECTION — block image theft from other domains
# ==================================================================
<IfModule mod_rewrite.c>
RewriteEngine On

RewriteCond %{HTTP_REFERER} !^$ [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?leadmax\.siteclap\.com [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?10projects\.com [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?google\. [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?bing\. [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?facebook\.com [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?twitter\.com [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?linkedin\.com [NC]
RewriteCond %{HTTP_REFERER} !^https?://(www\.)?pinterest\.com [NC]
RewriteCond %{HTTP_REFERER} !^https?://localhost [NC]
RewriteRule \.(jpg|jpeg|png|webp|avif|svg|gif)$ - [F,NC,L]
</IfModule>

# ==================================================================
# BOT / SCRAPER BLOCKING — block known scraper user agents
# ==================================================================
<IfModule mod_rewrite.c>
RewriteEngine On

# Block known scrapers on all pages (except API + assets)
RewriteCond %{HTTP_USER_AGENT} (HTTrack|SiteSucker|WebCopier|WebZIP|Teleport|Wget|libwww|Scrapy|python-requests|Go-http-client|Java/|MJ12bot|AhrefsBot|SemrushBot|DotBot|BLEXBot|DataForSeoBot|Bytespider|PetalBot) [NC]
RewriteCond %{REQUEST_URI} !^/wp-json/ [NC]
RewriteCond %{REQUEST_URI} !^/wp-admin/ [NC]
RewriteCond %{REQUEST_URI} !^/wp-content/ [NC]
RewriteCond %{REQUEST_URI} !^/wp-includes/ [NC]
RewriteRule ^ - [F,L]

# Block empty user agents on page requests (not assets)
RewriteCond %{HTTP_USER_AGENT} ^$
RewriteCond %{REQUEST_URI} !^/wp-content/ [NC]
RewriteCond %{REQUEST_URI} !^/wp-includes/ [NC]
RewriteCond %{REQUEST_URI} !^/wp-json/ [NC]
RewriteCond %{REQUEST_FILENAME} !-f
RewriteRule ^ - [F,L]
</IfModule>

# ==================================================================
# ROUTING — WordPress handles everything
# ==================================================================
RewriteEngine On
RewriteBase /

# ------------------------------------------------------------------
# 1. WordPress paths → let WordPress PHP handle them
# ------------------------------------------------------------------
RewriteRule ^wp-(admin|content|includes|login\.php|cron\.php|signup\.php) - [L]
RewriteRule ^xmlrpc\.php$ - [L]

# ------------------------------------------------------------------
# 2. WordPress REST API → rewrite to index.php for PHP handling
# ------------------------------------------------------------------
RewriteRule ^wp-json(/.*)?$ /index.php [L]

# ------------------------------------------------------------------
# 3. RankMath sitemaps → let WordPress handle
# ------------------------------------------------------------------
RewriteCond %{REQUEST_URI} sitemap [NC]
RewriteRule ^ - [L]

# robots.txt → let WordPress handle (RankMath manages it)
RewriteRule ^robots\.txt$ - [L]

# ------------------------------------------------------------------
# 4. Static files (CSS, JS, images, fonts) → serve directly
# ------------------------------------------------------------------
RewriteCond %{REQUEST_FILENAME} -f
RewriteRule ^ - [L]

# ------------------------------------------------------------------
# 5. Fallback → WordPress (all pages, 404s)
# ------------------------------------------------------------------
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule . /index.php [L]
